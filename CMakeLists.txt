cmake_minimum_required(VERSION 3.20)
project(orchard VERSION 0.1.0 LANGUAGES C)
option(ORCHARD_SANITIZE "Enable host address and undefined-behaviour sanitizers" OFF)
add_library(orchard STATIC
  src/machine.c src/cpu.c src/thumb.c src/mmu.c src/bus.c src/soc.c src/watchdog.c src/gpio.c src/interrupt.c src/rtc.c src/keyboard.c src/nor.c src/lcd.c src/adc.c src/usb.c)
target_include_directories(orchard PUBLIC include)
target_compile_features(orchard PUBLIC c_std_17)
set_target_properties(orchard PROPERTIES C_EXTENSIONS OFF)
add_executable(orchard-cli app/main.c)
target_link_libraries(orchard-cli PRIVATE orchard)
set_target_properties(orchard-cli PROPERTIES OUTPUT_NAME orchard C_EXTENSIONS OFF)
include(CTest)
set(orchard_checked_targets orchard orchard-cli)
if(BUILD_TESTING)
  add_executable(orchard-regression tests/regression.c)
  target_link_libraries(orchard-regression PRIVATE orchard)
  target_include_directories(orchard-regression PRIVATE src)
  set_target_properties(orchard-regression PROPERTIES C_EXTENSIONS OFF)
  add_test(NAME core-regression COMMAND orchard-regression)
  set_tests_properties(core-regression PROPERTIES TIMEOUT 30)
  add_executable(orchard-lcd-input tests/lcd-input.c)
  target_link_libraries(orchard-lcd-input PRIVATE orchard)
  target_include_directories(orchard-lcd-input PRIVATE src)
  set_target_properties(orchard-lcd-input PROPERTIES C_EXTENSIONS OFF)
  add_test(NAME lcd-input-regression COMMAND orchard-lcd-input)
  set_tests_properties(lcd-input-regression PROPERTIES TIMEOUT 30)
  list(APPEND orchard_checked_targets orchard-lcd-input)
  add_executable(orchard-lcd-adc tests/lcd-adc.c)
  target_link_libraries(orchard-lcd-adc PRIVATE orchard)
  target_include_directories(orchard-lcd-adc PRIVATE src)
  set_target_properties(orchard-lcd-adc PROPERTIES C_EXTENSIONS OFF)
  add_test(NAME lcd-adc-regression COMMAND orchard-lcd-adc)
  set_tests_properties(lcd-adc-regression PROPERTIES TIMEOUT 30)
  list(APPEND orchard_checked_targets orchard-lcd-adc)
  add_executable(orchard-usb tests/usb.c)
  target_link_libraries(orchard-usb PRIVATE orchard)
  target_include_directories(orchard-usb PRIVATE src)
  set_target_properties(orchard-usb PROPERTIES C_EXTENSIONS OFF)
  add_test(NAME usb-regression COMMAND orchard-usb)
  set_tests_properties(usb-regression PROPERTIES TIMEOUT 30)
  add_executable(orchard-usb-session tools/usb-session.c)
  target_link_libraries(orchard-usb-session PRIVATE orchard)
  set_target_properties(orchard-usb-session PROPERTIES C_EXTENSIONS OFF)
  list(APPEND orchard_checked_targets orchard-usb orchard-usb-session)
  # Opt-in, ROM-dependent research tool. Built but not run by CTest or installed.
  add_executable(orchard-gpio-window tools/probe-gpio-window.c)
  target_link_libraries(orchard-gpio-window PRIVATE orchard)
  target_include_directories(orchard-gpio-window PRIVATE src)
  set_target_properties(orchard-gpio-window PROPERTIES C_EXTENSIONS OFF)
  list(APPEND orchard_checked_targets orchard-regression orchard-gpio-window)
endif()
if(MSVC)
  foreach(target IN LISTS orchard_checked_targets)
    target_compile_options(${target} PRIVATE /W4 /WX)
  endforeach()
else()
  foreach(target IN LISTS orchard_checked_targets)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ORCHARD_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
install(TARGETS orchard orchard-cli)
install(FILES include/orchard/orchard.h DESTINATION include/orchard)
