cmake_minimum_required(VERSION 3.20)
project(orchard VERSION 0.1.0 LANGUAGES C)
option(ORCHARD_SANITIZE "Enable host address and undefined-behaviour sanitizers" OFF)
add_library(orchard STATIC
  src/machine.c src/cpu.c src/thumb.c src/mmu.c src/bus.c src/soc.c)
target_include_directories(orchard PUBLIC include)
target_compile_features(orchard PUBLIC c_std_17)
set_target_properties(orchard PROPERTIES C_EXTENSIONS OFF)
add_executable(orchard-cli app/main.c)
target_link_libraries(orchard-cli PRIVATE orchard)
set_target_properties(orchard-cli PROPERTIES OUTPUT_NAME orchard C_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(orchard PRIVATE /W4 /WX)
  target_compile_options(orchard-cli PRIVATE /W4 /WX)
else()
  foreach(target orchard orchard-cli)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ORCHARD_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
install(TARGETS orchard orchard-cli)
install(FILES include/orchard/orchard.h DESTINATION include/orchard)
